12/// The record schema this build reads and writes.
13pub const RECORD_VERSION: u64 = 1;
14/// Session names are tmux session names and file-name fragments; keep them short.
15pub const MAX_NAME_BYTES: usize = 128;
16/// Ids are uuids (36 bytes); the slack lets other minting schemes in without a format change.
17pub const MAX_ID_BYTES: usize = 64;
18
19/// A record id: a uuid v4 minted by `create_session`. Every way to build one validates the
20/// grammar `[A-Za-z0-9_-]{1,64}` (parsing, deserializing), so an id is always a safe file-name
21/// fragment and tmux option value: `<id>.json`, `<id>.lock`, `@tx-id`.
22#[derive(Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize, Deserialize)]
23#[serde(try_from = "String", into = "String")]
24pub struct SessionId(String);
25
26impl From<SessionId> for String {
27    fn from(id: SessionId) -> Self {
28        id.0
29    }
30}
31
32impl TryFrom<String> for SessionId {
33    type Error = InvalidSessionId;
34
35    fn try_from(id: String) -> Result<Self, Self::Error> {
36        id.parse()
37    }
38}
39
40impl SessionId {
41    pub fn mint() -> Self {
42        Self(uuid::Uuid::new_v4().to_string())
43    }
44
45    pub fn as_str(&self) -> &str {
46        &self.0
47    }
48}
49
50impl fmt::Display for SessionId {
51    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
52        f.write_str(&self.0)
53    }
54}

56#[derive(Debug, thiserror::Error)]
57#[error("invalid session id {0:?} (allowed: [A-Za-z0-9_-]+, at most {MAX_ID_BYTES} bytes)")]
58pub struct InvalidSessionId(pub String);
59
60impl FromStr for SessionId {
61    type Err = InvalidSessionId;
62
63    fn from_str(id: &str) -> Result<Self, Self::Err> {
64        if is_safe_fragment(id, MAX_ID_BYTES) {
65            Ok(Self(id.to_owned()))
66        } else {
67            Err(InvalidSessionId(id.to_owned()))
68        }
69    }
70}

72/// `[A-Za-z0-9_-]{1,max}`: safe as a path fragment, a tmux exact target and an option value.
73fn is_safe_fragment(text: &str, max: usize) -> bool {
74    !text.is_empty()
75        && text.len() <= max
76        && text
77            .bytes()
78            .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-')
79}

81#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
82#[serde(rename_all = "lowercase")]
83pub enum Lifecycle {
84    Running,
85    /// The tmux session vanished (process exit, server crash, a kill behind core's back).
86    Exited,
87    /// `stop_session` killed it.
88    Stopped,
89}
90
91impl Lifecycle {
92    pub const ALL: [Lifecycle; 3] = [Lifecycle::Running, Lifecycle::Exited, Lifecycle::Stopped];
93
94    /// Nothing in core moves a record out of a terminal state.
95    pub fn is_terminal(self) -> bool {
96        !matches!(self, Lifecycle::Running)
97    }
98
99    pub fn as_str(self) -> &'static str {
100        match self {
101            Lifecycle::Running => "running",
102            Lifecycle::Exited => "exited",
103            Lifecycle::Stopped => "stopped",
104        }
105    }
106}

108#[derive(Debug, thiserror::Error)]
109#[error("unknown lifecycle {0:?} (expected running, exited or stopped)")]
110pub struct InvalidLifecycle(pub String);
111
112impl FromStr for Lifecycle {
113    type Err = InvalidLifecycle;
114
115    fn from_str(text: &str) -> Result<Self, Self::Err> {
116        Self::ALL
117            .into_iter()
118            .find(|lifecycle| lifecycle.as_str() == text)
119            .ok_or_else(|| InvalidLifecycle(text.to_owned()))
120    }
121}

188/// The snapshot every read returns. Disk is the record; tmux holds only the `@tx-id` link.
189#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
190pub struct Session {
191    pub id: SessionId,
192    pub name: String,
193    /// Ancestry only; core never schedules or stops children.
194    pub parent: Option<SessionId>,
195    pub tags: Vec<String>,
196    pub cwd: PathBuf,
197    pub created_at: f64,
198    pub ended_at: Option<f64>,
199    pub lifecycle: Lifecycle,
200    pub launch: Launch,
201    /// Plugin-owned data keyed by plugin id; written through `TxCore::update_ext`. Core never
202    /// reads it.
203    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
204    pub ext: BTreeMap<String, serde_json::Value>,
205}

207#[derive(Debug, thiserror::Error)]
208pub enum RecordError {
209    #[error("{0}")]
210    Json(#[from] serde_json::Error),
211    #[error("record version {0} is not supported (this build reads version {RECORD_VERSION})")]
212    UnsupportedVersion(u64),
213    #[error("record version {0} is not an unsigned integer")]
214    InvalidVersion(String),
215    #[error("record has no version field")]
216    MissingVersion,
217    #[error("{0} is not a finite number")]
218    NonFiniteTime(&'static str),
219}
220
221#[derive(Serialize)]
222struct RecordOut<'a> {
223    version: u64,
224    #[serde(flatten)]
225    session: &'a Session,
226}
227
228#[derive(Deserialize)]
229struct RecordIn {
230    version: Option<u64>,
231    #[serde(flatten)]
232    session: Session,
233}
234
235impl Session {
236    /// Pretty JSON, newline-terminated, `version` first. Fails for what JSON cannot carry: a
237    /// non-UTF-8 `cwd` or a non-finite timestamp (serde_json would write `null`, which
238    /// `from_json` then refuses — better to refuse before the record exists).
239    pub fn to_json(&self) -> Result<String, RecordError> {
240        if !self.created_at.is_finite() {
241            return Err(RecordError::NonFiniteTime("created_at"));
242        }
243        if self.ended_at.is_some_and(|at| !at.is_finite()) {
244            return Err(RecordError::NonFiniteTime("ended_at"));
245        }
246        let record = RecordOut {
247            version: RECORD_VERSION,
248            session: self,
249        };
250        let mut text = serde_json::to_string_pretty(&record)?;
251        text.push('\n');
252        Ok(text)
253    }
254
255    pub fn from_json(text: &str) -> Result<Self, RecordError> {
256        let value: serde_json::Map<String, serde_json::Value> = serde_json::from_str(text)?;
257        match value.get("version") {
258            None => return Err(RecordError::MissingVersion),
259            Some(serde_json::Value::Number(version))
260                if version.as_u64() == Some(RECORD_VERSION) => {}
261            Some(serde_json::Value::Number(version)) => {
262                return Err(match version.as_u64() {
263                    Some(other) => RecordError::UnsupportedVersion(other),
264                    None => RecordError::InvalidVersion(version.to_string()),
265                });
266            }
267            Some(other) => return Err(RecordError::InvalidVersion(other.to_string())),
268        }
269        let record: RecordIn = serde_json::from_value(serde_json::Value::Object(value))?;
270        let _ = record.version;
271        Ok(record.session)
272    }
273
274    /// A plugin's own `ext` record, typed. `Ok(None)` when the plugin never wrote one; `Err`
275    /// when the stored value is not a `T`.
276    pub fn ext<T: serde::de::DeserializeOwned>(
277        &self,
278        plugin: &str,
279    ) -> Result<Option<T>, serde_json::Error> {
280        self.ext.get(plugin).map(T::deserialize).transpose()
281    }
282}

284/// `[A-Za-z0-9_-]+`, at most `MAX_NAME_BYTES`. The grammar keeps names safe as exact tmux targets
285/// (`=name`) and as path fragments.
286pub fn validate_name(name: &str) -> Result<(), CoreError> {
287    if is_safe_fragment(name, MAX_NAME_BYTES) {
288        Ok(())
289    } else {
290        Err(CoreError::InvalidName(name.to_owned()))
291    }
292}
