1//! Advisory file locks under `<state_dir>/locks/`: `create.lock` serializes name checks and
2//! `<id>.lock` serializes writes to (and keystrokes into) one session. `std::fs::File::lock`, no
3//! extra dependency. Lock order is always `create.lock` before `<id>.lock`.
4
5use std::fs::{File, OpenOptions};
6use std::io;
7use std::path::{Path, PathBuf};
8
9use crate::session::SessionId;
10
11pub const CREATE_LOCK: &str = "create.lock";
12const LOCK_SUFFIX: &str = ".lock";
13
14#[derive(Clone, Debug, PartialEq, Eq)]
15pub struct Locks {
16    dir: PathBuf,
17}
18
19/// An exclusive lock, released on drop.
20#[derive(Debug)]
21pub struct LockGuard {
22    file: File,
23    path: PathBuf,
24}
25
26impl LockGuard {
27    /// Whether the lock file at the path is still the inode this guard holds. Lock files are
28    /// never removed by core; if something else removes one, a holder cannot be protected (a
29    /// fresh file is a fresh lock), but an acquirer can at least refuse to settle on a ghost.
30    pub fn is_current(&self) -> io::Result<bool> {
31        use std::os::unix::fs::MetadataExt;
32        let held = self.file.metadata()?;
33        match std::fs::metadata(&self.path) {
34            Ok(at_path) => Ok(at_path.dev() == held.dev() && at_path.ino() == held.ino()),
35            Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(false),
36            Err(error) => Err(error),
37        }
38    }
39}
40
41impl Drop for LockGuard {
42    fn drop(&mut self) {
43        // Closing the descriptor releases the lock anyway; an unlock error has nowhere to go.
44        let _ = self.file.unlock();
45    }
46}
47
48impl Locks {
49    pub fn new(dir: impl Into<PathBuf>) -> Self {
50        Self { dir: dir.into() }
51    }
52
53    pub fn dir(&self) -> &Path {
54        &self.dir
55    }
56
57    /// Serializes name checks against session creation and renames.
58    pub fn create(&self) -> io::Result<LockGuard> {
59        self.acquire(&self.dir.join(CREATE_LOCK))
60    }
61
62    /// Serializes record writes and keystroke delivery for one session.
63    pub fn session(&self, id: &SessionId) -> io::Result<LockGuard> {
64        self.acquire(&self.dir.join(format!("{id}{LOCK_SUFFIX}")))
65    }
66
67    /// Open, lock, then confirm the path still names the locked inode; otherwise the file was
68    /// replaced between open and lock and the guard would be exclusive on a ghost — retry.
69    fn acquire(&self, path: &Path) -> io::Result<LockGuard> {
70        std::fs::create_dir_all(&self.dir)?;
71        const ATTEMPTS: usize = 8;
72        for _ in 0..ATTEMPTS {
73            let file = OpenOptions::new()
74                .read(true)
75                .write(true)
76                .create(true)
77                .truncate(false)
78                .open(path)?;
79            file.lock()?;
80            let guard = LockGuard {
81                file,
82                path: path.to_owned(),
83            };
84            if guard.is_current()? {
85                return Ok(guard);
86            }
87        }
88        Err(io::Error::other(format!(
89            "lock file {} keeps being replaced",
90            path.display()
91        )))
92    }
93}

95#[cfg(test)]
96mod tests {
97    use super::*;
98
99    fn sid(id: &str) -> SessionId {
100        id.parse().unwrap()
101    }
102    use std::os::unix::fs::PermissionsExt;
103    use std::sync::mpsc;
104    use std::time::{Duration, Instant};
105
106    #[test]
107    fn lock_files_live_under_the_dir() {
108        let dir = tempfile::tempdir().unwrap();
109        let locks = Locks::new(dir.path().join("locks"));
110        let _create = locks.create().unwrap();
111        let _session = locks.session(&sid("abc")).unwrap();
112        assert!(dir.path().join("locks/create.lock").is_file());
113        assert!(dir.path().join("locks/abc.lock").is_file());
114    }
115
116    #[test]
117    fn same_lock_serializes_two_threads() {
118        let dir = tempfile::tempdir().unwrap();
119        let locks = Locks::new(dir.path().join("locks"));
120        let (tx, rx) = mpsc::channel();
121        let guard = locks.create().unwrap();
122        let other = Locks::new(dir.path().join("locks"));
123        let waiter = std::thread::spawn(move || {
124            let started = Instant::now();
125            let _guard = other.create().unwrap();
126            tx.send(started.elapsed()).unwrap();
127        });
128        std::thread::sleep(Duration::from_millis(200));
129        drop(guard);
130        let waited = rx.recv().unwrap();
131        waiter.join().unwrap();
132        assert!(waited >= Duration::from_millis(150), "{waited:?}");
133    }
134
135    #[test]
136    fn different_session_locks_do_not_block() {
137        let dir = tempfile::tempdir().unwrap();
138        let locks = Locks::new(dir.path().join("locks"));
139        let _a = locks.session(&sid("a")).unwrap();
140        let started = Instant::now();
141        let _b = locks.session(&sid("b")).unwrap();
142        let _create = locks.create().unwrap();
143        assert!(started.elapsed() < Duration::from_millis(100));
144    }
145
146    #[test]
147    fn dropping_the_guard_releases_the_lock() {
148        let dir = tempfile::tempdir().unwrap();
149        let locks = Locks::new(dir.path().join("locks"));
150        drop(locks.session(&sid("a")).unwrap());
151        let started = Instant::now();
152        let _again = locks.session(&sid("a")).unwrap();
153        assert!(started.elapsed() < Duration::from_millis(100));
154    }
155
156    #[test]
157    fn a_lock_file_replaced_under_an_acquirer_is_reacquired_on_the_new_inode() {
158        let dir = tempfile::tempdir().unwrap();
159        let locks = Locks::new(dir.path().join("locks"));
160        let path = dir.path().join("locks/create.lock");
161        drop(locks.create().unwrap());
162        std::fs::remove_file(&path).unwrap();
163        let guard = locks.create().unwrap();
164        assert!(path.is_file());
165        assert!(guard.is_current().unwrap());
166        std::fs::remove_file(&path).unwrap();
167        assert!(!guard.is_current().unwrap());
168    }
169
170    #[test]
171    fn locks_dir_that_is_a_file_is_an_error() {
172        let dir = tempfile::tempdir().unwrap();
173        let file = dir.path().join("locks");
174        std::fs::write(&file, "").unwrap();
175        let locks = Locks::new(&file);
176        assert!(locks.create().is_err());
177        assert!(locks.session(&sid("a")).is_err());
178    }

180    #[test]
181    fn read_only_locks_dir_is_an_error() {
182        if unsafe_is_root() {
183            return;
184        }
185        let dir = tempfile::tempdir().unwrap();
186        let path = dir.path().join("locks");
187        std::fs::create_dir(&path).unwrap();
188        std::fs::set_permissions(&path, std::fs::Permissions::from_mode(0o500)).unwrap();
189        let locks = Locks::new(&path);
190        assert!(locks.create().is_err());
191        std::fs::set_permissions(&path, std::fs::Permissions::from_mode(0o700)).unwrap();
192    }

194    fn unsafe_is_root() -> bool {
195        std::fs::metadata("/").is_ok_and(|_| {
196            std::process::Command::new("id")
197                .arg("-u")
198                .output()
199                .is_ok_and(|out| String::from_utf8_lossy(&out.stdout).trim() == "0")
200        })
201    }
