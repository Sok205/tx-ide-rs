1//! The plugin host. One trait, [`Plugin`]; a [`Registry`] with four slots (engines, event
2//! sinks, verbs, resources); a [`PluginConfig`] the front end fills from its own config format.
3//! Core never parses arguments or reads the environment: a [`Verb`] receives argv already split.
4//! A resource is a named static blob the core stores and lists but never reads.
5

18pub trait Plugin: Send + Sync {
19    /// `[a-z0-9-]+`, unique across the manifest; the key in the front end's config and in
20    /// `Session.ext`.
21    fn id(&self) -> &'static str;
22    fn register(&self, r: &mut Registry) -> Result<(), PluginError>;
23}
24
25/// What one plugin contributes. Built per plugin by [`TxCore::plug`].
26pub struct Registry {
27    engines: Vec<Box<dyn EngineAdapter>>,
28    sinks: Vec<Box<dyn EventSink>>,
29    verbs: Vec<Arc<dyn Verb>>,
30    resources: BTreeMap<&'static str, &'static [u8]>,
31    settings: Value,
32    config: CoreConfig,
33}
34
35impl Registry {
36    pub fn engine(&mut self, adapter: Box<dyn EngineAdapter>) {
37        self.engines.push(adapter);
38    }
39
40    pub fn sink(&mut self, sink: Box<dyn EventSink>) {
41        self.sinks.push(sink);
42    }
43
44    pub fn verb(&mut self, verb: Arc<dyn Verb>) {
45        self.verbs.push(verb);
46    }
47
48    /// Publishes a named static blob; the core stores and lists it, never reads it. `name` is
49    /// `[A-Za-z0-9_./-]+`, no leading `/`, no `..` segment, unique across plugins. Convention:
50    /// `<kind>/<file>`, e.g. `keys/kill.tmux`.
51    pub fn resource(
52        &mut self,
53        name: &'static str,
54        bytes: &'static [u8],
55    ) -> Result<(), PluginError> {
56        validate_resource_name(name)?;
57        if self.resources.insert(name, bytes).is_some() {
58            return Err(PluginError::DuplicateResource(name.to_owned()));
59        }
60        Ok(())
61    }
62
63    /// This plugin's settings block; `Null` when the config has none.
64    pub fn settings(&self) -> &Value {
65        &self.settings
66    }
67
68    /// The core's config (state dir, socket), for plugins that open a second core in a sink.
69    pub fn config(&self) -> &CoreConfig {
70        &self.config
71    }
72}

82/// A command a front end can dispatch to. `argv` is everything after the verb name.
83pub trait Verb: Send + Sync {
84    fn name(&self) -> &'static str;
85    /// One usage line, shown under the front end's own verbs.
86    fn usage(&self) -> &'static str;
87    /// Returns the exit code. Usage mistakes: write to `io.stderr`, return `Ok(2)`.
88    fn run(&self, core: &TxCore, io: &mut VerbIo<'_>, argv: &[String]) -> Result<i32, VerbError>;
89}

130#[derive(Debug, thiserror::Error)]
131#[non_exhaustive]
132pub enum PluginError {
133    #[error("invalid plugin id {0:?} (allowed: [a-z0-9-]+)")]
134    InvalidId(String),
135    #[error("plugin {0:?} is listed twice")]
136    DuplicatePlugin(String),
137    #[error("no plugin named {0:?}")]
138    Unknown(String),
139    #[error("verb {verb:?} is registered by both {first:?} and {second:?}")]
140    DuplicateVerb {
141        verb: String,
142        first: String,
143        second: String,
144    },
145    #[error("plugin {plugin:?} registers verb {verb:?}, which the front end owns")]
146    ReservedVerb { verb: String, plugin: String },
147    #[error("resource {0:?} is published twice")]
148    DuplicateResource(String),
149    #[error(
150        "invalid resource name {0:?} (allowed: [A-Za-z0-9_./-]+, no leading '/', no '..' segment)"
151    )]
152    InvalidResourceName(String),
153    #[error("plugin {plugin:?}: {source}")]
154    Engine { plugin: String, source: CoreError },
155    #[error("plugin {plugin:?}: {reason}")]
156    Register { plugin: String, reason: String },
157}

159fn validate_id(id: &str) -> Result<(), PluginError> {
160    let ok = !id.is_empty()
161        && id
162            .bytes()
163            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-');
164    if ok {
165        Ok(())
166    } else {
167        Err(PluginError::InvalidId(id.to_owned()))
168    }
169}
170
171fn validate_resource_name(name: &str) -> Result<(), PluginError> {
172    let ok = !name.is_empty()
173        && !name.starts_with('/')
174        && name
175            .bytes()
176            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'_' | b'.' | b'/' | b'-'))
177        && name.split('/').all(|segment| segment != "..");
178    if ok {
179        Ok(())
180    } else {
181        Err(PluginError::InvalidResourceName(name.to_owned()))
182    }
183}

185/// The host side, kept in `TxCore`.
186#[derive(Default)]
187pub(crate) struct Host {
188    /// name → (plugin id, verb)
189    verbs: BTreeMap<String, (String, Arc<dyn Verb>)>,
190    /// Manifest order, then name order within a plugin.
191    resources: Vec<Resource>,
192    status: Vec<(String, PluginStatus)>,
193}

195/// One plugin after `register`: `None` when the config disabled it.
196struct Loaded {
197    id: String,
198    registry: Option<Registry>,
199}
200
201/// Validates ids and config, runs every enabled plugin's `register` against a fresh
202/// [`Registry`]. Touches nothing outside memory.
203fn load(
204    plugins: &[Box<dyn Plugin>],
205    cfg: &PluginConfig,
206    config: &CoreConfig,
207) -> Result<Vec<Loaded>, PluginError> {
208    let ids: BTreeSet<&str> = plugins.iter().map(|p| p.id()).collect();
209    for id in cfg.disabled.iter().chain(cfg.settings.keys()) {
210        if !ids.contains(id.as_str()) {
211            return Err(PluginError::Unknown(id.clone()));
212        }
213    }
214    let mut seen = BTreeSet::new();
215    let mut loaded = Vec::with_capacity(plugins.len());
216    for plugin in plugins {
217        let id = plugin.id();
218        validate_id(id)?;
219        if !seen.insert(id) {
220            return Err(PluginError::DuplicatePlugin(id.to_owned()));
221        }
222        if cfg.disabled.contains(id) {
223            loaded.push(Loaded {
224                id: id.to_owned(),
225                registry: None,
226            });
227            continue;
228        }
229        let mut registry = Registry {
230            engines: Vec::new(),
231            sinks: Vec::new(),
232            verbs: Vec::new(),
233            resources: BTreeMap::new(),
234            settings: cfg.settings.get(id).cloned().unwrap_or(Value::Null),
235            config: config.clone(),
236        };
237        plugin.register(&mut registry)?;
238        loaded.push(Loaded {
239            id: id.to_owned(),
240            registry: Some(registry),
241        });
242    }
243    Ok(loaded)
244}

246type VerbTable = BTreeMap<String, (String, Arc<dyn Verb>)>;
247
248fn insert_verb(
249    table: &mut VerbTable,
250    cfg: &PluginConfig,
251    plugin: &str,
252    verb: Arc<dyn Verb>,
253) -> Result<(), PluginError> {
254    let name = verb.name().to_owned();
255    if cfg.reserved.contains(&name) {
256        return Err(PluginError::ReservedVerb {
257            verb: name,
258            plugin: plugin.to_owned(),
259        });
260    }
261    if let Some((first, _)) = table.get(&name) {
262        return Err(PluginError::DuplicateVerb {
263            verb: name,
264            first: first.clone(),
265            second: plugin.to_owned(),
266        });
267    }
268    table.insert(name, (plugin.to_owned(), verb));
269    Ok(())
270}

272/// The verbs [`TxCore::plug`] would expose for this manifest and config, without a core: for
273/// usage text and "is this verb known?" checks that must not create the state dir. Runs every
274/// enabled plugin's `register`; engines and sinks are dropped.
275pub fn preview_verbs(
276    plugins: &[Box<dyn Plugin>],
277    cfg: &PluginConfig,
278    config: &CoreConfig,
279) -> Result<Vec<Arc<dyn Verb>>, PluginError> {
280    let mut table = VerbTable::new();
281    for loaded in load(plugins, cfg, config)? {
282        if let Some(registry) = loaded.registry {
283            for verb in registry.verbs {
284                insert_verb(&mut table, cfg, &loaded.id, verb)?;
285            }
286        }
287    }
288    Ok(table.into_values().map(|(_, verb)| verb).collect())
289}

291impl TxCore {
292    /// Registers every plugin not in `cfg.disabled`, in order. Fails on the first problem; the
293    /// core may then hold a partial registration and the caller should exit.
294    pub fn plug(
295        &mut self,
296        plugins: Vec<Box<dyn Plugin>>,
297        cfg: &PluginConfig,
298    ) -> Result<(), PluginError> {
299        for loaded in load(&plugins, cfg, self.config())? {
300            let Some(registry) = loaded.registry else {
301                self.host.status.push((loaded.id, PluginStatus::Disabled));
302                continue;
303            };
304            for adapter in registry.engines {
305                self.register_engine(adapter)
306                    .map_err(|source| PluginError::Engine {
307                        plugin: loaded.id.clone(),
308                        source,
309                    })?;
310            }
311            for sink in registry.sinks {
312                self.subscribe(sink);
313            }
314            for verb in registry.verbs {
315                insert_verb(&mut self.host.verbs, cfg, &loaded.id, verb)?;
316            }
317            for (name, bytes) in registry.resources {
318                if self.host.resources.iter().any(|r| r.name == name) {
319                    return Err(PluginError::DuplicateResource(name.to_owned()));
320                }
321                self.host.resources.push(Resource {
322                    plugin: loaded.id.clone(),
323                    name: name.to_owned(),
324                    bytes,
325                });
326            }
327            self.host.status.push((loaded.id, PluginStatus::Active));
328        }
329        Ok(())
330    }
331
332    pub fn verb(&self, name: &str) -> Option<Arc<dyn Verb>> {
333        self.host.verbs.get(name).map(|(_, verb)| Arc::clone(verb))
334    }
335
336    /// Every plugin verb, by name.
337    pub fn verbs(&self) -> Vec<Arc<dyn Verb>> {
338        self.host
339            .verbs
340            .values()
341            .map(|(_, verb)| Arc::clone(verb))
342            .collect()
343    }
344
345    /// Resources of active plugins whose name starts with `prefix`, in manifest order then name
346    /// order.
347    pub fn resources(&self, prefix: &str) -> Vec<Resource> {
348        self.host
349            .resources
350            .iter()
351            .filter(|r| r.name.starts_with(prefix))
352            .cloned()
353            .collect()
354    }
355
356    /// Manifest order.
357    pub fn plugin_status(&self) -> &[(String, PluginStatus)] {
358        &self.host.status
359    }
