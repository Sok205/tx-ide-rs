1//! Events: what happened to one session. Every mutation produces [`Event`]s that go to the
2//! operation log (`<state_dir>/log.jsonl`, one compact JSON line
3//! `{"ts","actor","type","id","name","msg"}` per event, appended with a single write on an
4//! `O_APPEND` descriptor so concurrent writers never tear a line) and then to every in-process
5//! [`EventSink`].
6

17/// The smallest `PIPE_BUF` among supported platforms (macOS); a line at most this long is
18/// appended in one atomic write. The newline counts.
19pub const PIPE_BUF: usize = 512;
20
21/// Every event type core logs.
22#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
23#[non_exhaustive]
24pub enum EventType {
25    Created,
26    Renamed,
27    Tagged,
28    MessageSent,
29    Exited,
30    Stopped,
31    Forgotten,
32}
33
34impl EventType {
35    pub const ALL: &'static [EventType] = &[
36        EventType::Created,
37        EventType::Renamed,
38        EventType::Tagged,
39        EventType::MessageSent,
40        EventType::Exited,
41        EventType::Stopped,
42        EventType::Forgotten,
43    ];
44
45    pub fn as_str(self) -> &'static str {
46        match self {
47            EventType::Created => "created",
48            EventType::Renamed => "renamed",
49            EventType::Tagged => "tagged",
50            EventType::MessageSent => "message-sent",
51            EventType::Exited => "exited",
52            EventType::Stopped => "stopped",
53            EventType::Forgotten => "forgotten",
54        }
55    }
56}

58/// One thing that happened to one session. Handed to in-process sinks and written to the log.
59#[derive(Clone, Debug, PartialEq)]
60#[non_exhaustive]
61pub struct Event {
62    pub ts: f64,
63    pub type_: EventType,
64    pub id: SessionId,
65    pub name: String,
66    /// The sending session for `message-sent`; `None` means the user or core itself.
67    pub actor: Option<SessionId>,
68    /// Human-readable detail (the old name on a rename, the tag list, …). Not a contract.
69    pub detail: String,
70}
71
72impl Event {
73    pub fn new(
74        type_: EventType,
75        id: SessionId,
76        name: impl Into<String>,
77        actor: Option<SessionId>,
78        detail: impl Into<String>,
79    ) -> Self {
80        Self {
81            ts: now(),
82            type_,
83            id,
84            name: name.into(),
85            actor,
86            detail: detail.into(),
87        }
88    }
89}

91/// An in-process listener. Called after the operation's record is saved and every lock is
92/// released, in registration order; cannot veto; never receives a handle to the core (open a
93/// second `TxCore` on the same state dir if you need one). Hand slow work to your own thread.
94///
95/// Order is per core: another process's events reach you only through the log, and log order
96/// across processes is write order, not causal order.
97///
98/// A panicking sink is contained with `catch_unwind`: the operation already happened and the
99/// log line is already written. The default panic hook still prints the panic to stderr (core
100/// installs no hook; that is the front end's call), the sink's own `Mutex`es are poisoned
101/// afterwards, and nothing is contained under `panic = "abort"`.
102///
103/// Closures (`|e: &Event| …`) and `Arc<impl EventSink>` are sinks. The closure impl rules out a
104/// future blanket impl for `Box<T>` or `&T`; box your sink when registering it.
105pub trait EventSink: Send + Sync {
106    fn on_event(&self, event: &Event);
107}
108
109impl<F: Fn(&Event) + Send + Sync> EventSink for F {
110    fn on_event(&self, event: &Event) {
111        self(event)
112    }
113}
114
115impl<T: EventSink + ?Sized> EventSink for std::sync::Arc<T> {
116    fn on_event(&self, event: &Event) {
117        (**self).on_event(event)
118    }
119}

121#[derive(Clone, Debug, PartialEq, Eq)]
122pub struct EventLog {
123    path: PathBuf,
124}
125
126impl EventLog {
127    pub fn new(path: impl Into<PathBuf>) -> Self {
128        Self { path: path.into() }
129    }
130
131    pub fn path(&self) -> &Path {
132        &self.path
133    }
134
135    /// Append one event as one line.
136    pub fn append(&self, event: &Event) -> io::Result<()> {
137        self.append_at(
138            event.ts,
139            event.type_.as_str(),
140            event.id.as_str(),
141            &event.name,
142            &event.detail,
143            event.actor.as_ref().map_or("", SessionId::as_str),
144        )
145    }
146
147    pub fn append_at(
148        &self,
149        ts: f64,
150        type_: &str,
151        id: &str,
152        name: &str,
153        msg: &str,
154        actor: &str,
155    ) -> io::Result<()> {
156        let line = encode(ts, actor, type_, id, name, msg);
157        if let Some(parent) = self.path.parent() {
158            std::fs::create_dir_all(parent)?;
159        }
160        let mut file = OpenOptions::new()
161            .append(true)
162            .create(true)
163            .mode(0o644)
164            .open(&self.path)?;
165        file.write_all(&line)
166    }
167}

169/// Seconds since the epoch as a float.
170pub fn now() -> f64 {
171    let nanos = SystemTime::now()
172        .duration_since(UNIX_EPOCH)
173        .map_or(0, |elapsed| elapsed.as_nanos());
174    nanos as f64 / 1e9
175}
176
177fn line(ts: f64, actor: &str, type_: &str, id: &str, name: &str, msg: &str) -> String {
178    let ts = Number::from_f64(ts).map_or(Value::Null, Value::Number);
179    let mut line = json!({
180        "ts": ts, "actor": actor, "type": type_, "id": id, "name": name, "msg": msg
181    })
182    .to_string();
183    line.push('\n');
184    line
185}

187/// Compact JSON + newline, `msg` trimmed on a char boundary so the line fits `PIPE_BUF`.
188pub fn encode(ts: f64, actor: &str, type_: &str, id: &str, name: &str, msg: &str) -> Vec<u8> {
189    let full = line(ts, actor, type_, id, name, msg);
190    if full.len() <= PIPE_BUF {
191        return full.into_bytes();
192    }
193    let skeleton = line(ts, actor, type_, id, name, "").len();
194    let mut budget = PIPE_BUF.saturating_sub(skeleton);
195    let mut end = 0;
196    for (index, c) in msg.char_indices() {
197        let escaped = escaped_len(c);
198        if escaped > budget {
199            break;
200        }
201        budget -= escaped;
202        end = index + c.len_utf8();
203    }
204    line(ts, actor, type_, id, name, &msg[..end]).into_bytes()
205}
206
207/// Bytes `c` occupies inside a serde_json string: short escapes are 2 bytes, other C0 controls
208/// are `\u00XX`, everything else is its UTF-8 length.
209fn escaped_len(c: char) -> usize {
210    match c {
211        '"' | '\\' | '\n' | '\r' | '\t' | '\u{8}' | '\u{c}' => 2,
212        '\0'..='\u{1f}' => 6,
213        c => c.len_utf8(),
214    }
215}

243    #[test]
244    fn ascii_overflow_is_trimmed_to_exactly_pipe_buf() {
245        let msg = format!("{} (r2)", "y".repeat(1000));
246        let line = encode(1727000000.1234567, "sess-1", "renamed", "i", "n", &msg);
247        assert_eq!(line.len(), PIPE_BUF);
248        let value = parse(&line);

252    #[test]
253    fn exact_pipe_buf_is_kept_and_one_over_loses_one_char() {
254        let skeleton = encode(0.0, "sess-1", "t", "i", "n", "").len();
255        let fits = "y".repeat(PIPE_BUF - skeleton);
256        let line = encode(0.0, "sess-1", "t", "i", "n", &fits);
257        assert_eq!(line.len(), PIPE_BUF);
258        assert_eq!(parse(&line)["msg"], fits);
259        let over = "y".repeat(PIPE_BUF + 1 - skeleton);
260        let trimmed = encode(0.0, "sess-1", "t", "i", "n", &over);
261        assert_eq!(trimmed, line);
262    }

264    #[test]
265    fn multibyte_and_escaped_chars_stay_within_pipe_buf() {
266        for msg in [
267            "é".repeat(600),
268            "😀".repeat(200),
269            "\"".repeat(600),
270            "\\".repeat(600),
271            "\n".repeat(600),
272            "\u{1}".repeat(600),

274            "a\u{1}é😀\"".repeat(100),
275        ] {
276            let line = encode(0.0, "a", "t", "i", "n", &msg);
277            assert!(line.len() <= PIPE_BUF, "{}", line.len());

341    #[test]
342    fn concurrent_appends_never_tear() {
343        let dir = tempfile::tempdir().unwrap();
344        let path = dir.path().join("log.jsonl");
345        let handles: Vec<_> = (0..4)
346            .map(|worker| {
347                let log = EventLog::new(&path);
348                std::thread::spawn(move || {
349                    let msg = format!("{worker}").repeat(400);
350                    for _ in 0..200 {
351                        log.append_at(0.0, "t", "i", "n", &msg, "").unwrap();
352                    }
353                })
354            })
355            .collect();
356        for handle in handles {
357            handle.join().unwrap();
358        }
359        let text = std::fs::read_to_string(&path).unwrap();
360        let lines: Vec<_> = text.lines().collect();
361        assert_eq!(lines.len(), 800);
362        for line in lines {
363            let value: Value = serde_json::from_str(line).unwrap();
364            let msg = value["msg"].as_str().unwrap();
365            assert!(
366                msg.chars().all(|c| c == msg.chars().next().unwrap()),
367                "torn: {msg}"
368            );
369        }
370    }
