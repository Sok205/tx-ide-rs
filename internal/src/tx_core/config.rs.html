1//! `CoreConfig`: everything the library needs, handed in by the caller. The library never reads
2//! the process environment.
3
4use std::path::PathBuf;
5
6use crate::error::CoreError;
7
8/// The longest tmux socket path `open` accepts. `sun_path` holds 104 bytes on macOS and 108 on
9/// Linux, and tmux fails opaquely near the limit; 100 leaves room for the NUL and some slack.
10pub const MAX_SOCKET_PATH_BYTES: usize = 100;
11
12pub const SOCKET_FILE: &str = "tmux.sock";
13pub const CONF_FILE: &str = "tmux.conf";
14
15#[derive(Clone, Debug, PartialEq, Eq)]
16#[non_exhaustive]
17pub struct CoreConfig {
18    /// Records, log, locks, profiles and (by default) the tmux socket live here.
19    pub state_dir: PathBuf,
20    /// Overrides `<state_dir>/tmux.sock` when that path would be too long.
21    pub tmux_socket: Option<PathBuf>,
22    /// Base environment for every session, applied before the per-session env.
23    pub env: Vec<(String, String)>,
24    /// Role search dirs, in order; a later dir overrides an earlier one.
25    pub profile_dirs: Vec<PathBuf>,
26    /// Skill search dirs, in order; a later dir overrides an earlier one.
27    pub skill_dirs: Vec<PathBuf>,
28}
29
30impl CoreConfig {
31    pub fn new(state_dir: impl Into<PathBuf>) -> Self {
32        Self {
33            state_dir: state_dir.into(),
34            tmux_socket: None,
35            env: Vec::new(),
36            profile_dirs: Vec::new(),
37            skill_dirs: Vec::new(),
38        }
39    }
40
41    pub fn tmux_socket(mut self, socket: impl Into<PathBuf>) -> Self {
42        self.tmux_socket = Some(socket.into());
43        self
44    }
45
46    pub fn env(mut self, key: impl Into<String>, value: impl Into<String>) -> Self {
47        self.env.push((key.into(), value.into()));
48        self
49    }
50
51    pub fn profile_dirs<I, P>(mut self, dirs: I) -> Self
52    where
53        I: IntoIterator<Item = P>,
54        P: Into<PathBuf>,
55    {
56        self.profile_dirs.extend(dirs.into_iter().map(Into::into));
57        self
58    }
59
60    pub fn skill_dirs<I, P>(mut self, dirs: I) -> Self
61    where
62        I: IntoIterator<Item = P>,
63        P: Into<PathBuf>,
64    {
65        self.skill_dirs.extend(dirs.into_iter().map(Into::into));
66        self
67    }
68
69    pub fn socket_path(&self) -> PathBuf {
70        self.tmux_socket
71            .clone()
72            .unwrap_or_else(|| self.state_dir.join(SOCKET_FILE))
73    }
74
75    pub fn tmux_conf_path(&self) -> PathBuf {
76        self.state_dir.join(CONF_FILE)
77    }
78
79    /// Fails before any file is touched when the socket path cannot fit a `sockaddr_un` or is
80    /// not absolute (a relative socket would move with the caller's cwd and be exported into
81    /// sessions as such).
82    pub fn check_socket_path(&self) -> Result<(), CoreError> {
83        let socket = self.socket_path();
84        if socket.as_os_str().len() > MAX_SOCKET_PATH_BYTES {
85            return Err(CoreError::SocketPathTooLong(socket));
86        }
87        if !socket.is_absolute() {
88            return Err(CoreError::SocketPathNotAbsolute(socket));
89        }
90        Ok(())
91    }
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97    use std::path::{Path, PathBuf};
98
99    #[test]
100    fn socket_defaults_under_state_dir() {
101        let config = CoreConfig::new("/tmp/txc");
102        assert_eq!(config.socket_path(), PathBuf::from("/tmp/txc/tmux.sock"));
103        assert_eq!(config.tmux_conf_path(), PathBuf::from("/tmp/txc/tmux.conf"));
104    }
105
106    #[test]
107    fn explicit_socket_wins() {
108        let mut config = CoreConfig::new("/tmp/txc");
109        config.tmux_socket = Some("/tmp/s.sock".into());
110        assert_eq!(config.socket_path(), PathBuf::from("/tmp/s.sock"));
111        assert_eq!(config.tmux_conf_path(), PathBuf::from("/tmp/txc/tmux.conf"));
112    }
113
114    #[test]
115    fn empty_by_default() {
116        let config = CoreConfig::new("/x");
117        assert!(config.env.is_empty());
118        assert!(config.profile_dirs.is_empty());
119        assert!(config.skill_dirs.is_empty());
120        assert_eq!(config.state_dir, Path::new("/x"));
121    }
122
123    /// A state dir whose `tmux.sock` lands at exactly `len` bytes.
124    fn state_dir_for_socket_len(len: usize) -> PathBuf {
125        let dir = format!("/{}", "d".repeat(len - "/".len() - "/tmux.sock".len()));
126        let config = CoreConfig::new(&dir);
127        assert_eq!(config.socket_path().as_os_str().len(), len);
128        PathBuf::from(dir)
129    }
130
131    #[test]
132    fn socket_path_of_100_bytes_is_ok() {
133        let config = CoreConfig::new(state_dir_for_socket_len(MAX_SOCKET_PATH_BYTES));
134        assert!(config.check_socket_path().is_ok());
135    }
136
137    #[test]
138    fn socket_path_of_101_bytes_is_too_long() {
139        let config = CoreConfig::new(state_dir_for_socket_len(MAX_SOCKET_PATH_BYTES + 1));
140        let error = config.check_socket_path().unwrap_err();
141        let CoreError::SocketPathTooLong(path) = &error else {
142            panic!("{error:?}")
143        };
144        assert_eq!(*path, config.socket_path());
145        assert!(error.to_string().contains("101 bytes"), "{error}");
146    }
147
148    #[test]
149    fn relative_or_empty_socket_is_refused() {
150        for bad in ["", "rel.sock", "."] {
151            let config = CoreConfig::new("/abs").tmux_socket(bad);
152            assert!(
153                matches!(
154                    config.check_socket_path(),
155                    Err(CoreError::SocketPathNotAbsolute(_))
156                ),
157                "{bad:?}"
158            );
159        }
160        assert!(CoreConfig::new("rel").check_socket_path().is_err());
161    }
162
163    #[test]
164    fn explicit_long_socket_is_checked_too() {
165        let mut config = CoreConfig::new("/short");
166        config.tmux_socket = Some(PathBuf::from(format!("/{}", "s".repeat(120))));
167        assert!(matches!(
168            config.check_socket_path(),
169            Err(CoreError::SocketPathTooLong(_))
170        ));
171    }
172}
