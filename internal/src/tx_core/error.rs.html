1//! The one error type every public operation returns.
2
3use std::io;
4use std::path::PathBuf;
5
6use crate::engine::EngineError;
7use crate::profile::RoleError;
8use crate::session::SessionId;
9use crate::store::StoreError;
10use crate::tmux::TmuxError;
11
12#[derive(Debug, thiserror::Error)]
13#[non_exhaustive]
14pub enum CoreError {
15    #[error("no session {0:?}")]
16    NotFound(String),
17    #[error("a running session is already named {0:?}")]
18    NameTaken(String),
19    #[error("invalid session name {0:?} (allowed: [A-Za-z0-9_-]+, at most {max} bytes)", max = crate::session::MAX_NAME_BYTES)]
20    InvalidName(String),
21    #[error("invalid environment key {0:?} (a name without '=' or NUL)")]
22    InvalidEnvKey(String),
23    #[error("{} is not a directory", .0.display())]
24    NotADirectory(PathBuf),
25    #[error("session {0} is not running")]
26    NotRunning(SessionId),
27    #[error("no engine registered as {0:?}")]
28    UnknownEngine(String),
29    #[error("an engine with id {0:?} is already registered")]
30    EngineTaken(String),
31    #[error("tmux socket path is {len} bytes, longer than the {max} allowed: {path}", len = .0.as_os_str().len(), max = crate::config::MAX_SOCKET_PATH_BYTES, path = .0.display())]
32    SocketPathTooLong(PathBuf),
33    #[error("tmux socket path must be absolute: {}", .0.display())]
34    SocketPathNotAbsolute(PathBuf),
35    #[error(transparent)]
36    Profile(#[from] RoleError),
37    #[error(transparent)]
38    Engine(#[from] EngineError),
39    #[error(transparent)]
40    Store(#[from] StoreError),
41    #[error(transparent)]
42    Tmux(#[from] TmuxError),
43    #[error("lock: {0}")]
44    Lock(io::Error),
45}
