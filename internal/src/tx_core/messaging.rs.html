1//! Delivery of one message to one running session: the envelope is typed into the pane with
2//! `send-keys -l`, a pause lets the engine's input handling settle, then `Enter` submits it. A
3//! receipt means input was submitted: not read, not completed, not retried.
4
5use std::time::Duration;
6
7use crate::tmux::{Tmux, TmuxError};
8
9/// Between typing the envelope and pressing `Enter`: long enough for an interactive engine to
10/// register pasted-looking input as text rather than a shortcut.
11pub const DELIVERY_PAUSE: Duration = Duration::from_millis(300);
12
13/// `sender: None` is the user; `Some(name)` is another agent, named by its session name.
14pub fn build_envelope(sender: Option<&str>, text: &str) -> String {
15    match sender {
16        None => format!("<from-user>{text}</from-user>"),
17        Some(name) => format!("<from-agent session=\"{name}\">{text}</from-agent>"),
18    }
19}
20
21/// Types `envelope` into `pane`, waits `pause`, submits with `Enter`.
22pub fn deliver(tmux: &Tmux, pane: &str, envelope: &str, pause: Duration) -> Result<(), TmuxError> {
23    tmux.send_keys_literal(pane, envelope)?;
24    std::thread::sleep(pause);
25    tmux.send_enter(pane)
26}

28#[cfg(test)]
29mod tests {
30    use super::*;
31    use std::path::Path;
32
33    #[test]
34    fn user_envelope() {
35        assert_eq!(build_envelope(None, "hi"), "<from-user>hi</from-user>");
36    }
37
38    #[test]
39    fn agent_envelope_names_the_sender() {
40        assert_eq!(
41            build_envelope(Some("worker-1"), "hi"),
42            "<from-agent session=\"worker-1\">hi</from-agent>"
43        );
44    }
45
46    #[test]
47    fn text_is_kept_verbatim() {
48        let text = "line one\nline two; <tag> \"q\" ; é 😀";
49        assert_eq!(
50            build_envelope(None, text),
51            format!("<from-user>{text}</from-user>")
52        );
53        assert_eq!(build_envelope(None, ""), "<from-user></from-user>");
54    }
55
56    #[test]
57    fn pause_is_the_documented_value() {
58        assert_eq!(DELIVERY_PAUSE, Duration::from_millis(300));
59    }
60
61    #[test]
62    fn deliver_types_the_envelope_and_submits_it() {
63        use crate::tmux::Tmux;
64        use std::process::Command;
65        if Command::new("tmux").arg("-V").output().is_err() {
66            eprintln!("tmux not available; skipping");
67            return;
68        }
69        let dir = tempfile::Builder::new()
70            .prefix("txc")
71            .tempdir_in("/tmp")
72            .unwrap();
73        let tmux = Tmux::new("tmux", dir.path().join("tmux.sock"), "/dev/null");
74        let out = dir.path().join("got.txt");
75        let id: crate::session::SessionId = "m".parse().unwrap();
76        tmux.new_session(
77            "m",
78            Path::new("/"),
79            "sh",
80            &[
81                "-c".into(),
82                format!("head -n 2 > {}; sleep 30", out.display()),
83            ],
84            &[],
85        )
86        .unwrap();
87        tmux.set_tx_id("m", &id).unwrap();
88        let pane = tmux.is_live(&id).unwrap().unwrap().pane_target;
89        std::thread::sleep(Duration::from_millis(200));
90        deliver(&tmux, &pane, "-x ; one;", Duration::from_millis(10)).unwrap();
91        deliver(&tmux, &pane, "two", Duration::from_millis(10)).unwrap();
92        for _ in 0..50 {
93            if out.exists() && std::fs::read_to_string(&out).unwrap().lines().count() == 2 {
94                break;
95            }
96            std::thread::sleep(Duration::from_millis(100));
97        }
98        assert_eq!(std::fs::read_to_string(&out).unwrap(), "-x ; one;\ntwo\n");
99        assert!(deliver(&tmux, "%999", "x", Duration::ZERO).is_err());
100        let _ = Command::new("tmux")
101            .args(tmux.argv(&["kill-server".into()]))
102            .output();
103    }
104}
