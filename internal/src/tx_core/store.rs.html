1//! `SessionStore`: `<dir>/<id>.json`, one file per record, written atomically (temp file in the
2//! same directory, then rename) so a reader never sees a half-written record.
3
4use std::io::{self, Write};
5use std::path::{Path, PathBuf};
6
7use crate::session::{RecordError, Session, SessionId};
8
9const RECORD_SUFFIX: &str = ".json";
10const TEMP_SUFFIX: &str = ".tmp";
11
12/// Why one record file could not be read.
13#[derive(Debug, thiserror::Error)]
14pub enum ReadError {
15    #[error("{0}")]
16    Io(#[from] io::Error),
17    #[error("not UTF-8: {0}")]
18    Utf8(std::str::Utf8Error),
19    #[error("{0}")]
20    Record(#[from] RecordError),
21    #[error("not a regular file")]
22    NotAFile,
23    #[error("record carries id {found:?}, which is not the file's name")]
24    IdMismatch { found: String },
25}
26
27#[derive(Debug, thiserror::Error)]
28pub enum StoreError {
29    #[error("{}: {source}", path.display())]
30    Io {
31        path: PathBuf,
32        #[source]
33        source: io::Error,
34    },
35    /// A record file exists but cannot be read. Named so a human can inspect or remove it.
36    #[error("unreadable record {file}: {source}")]
37    Unreadable {
38        file: String,
39        #[source]
40        source: ReadError,
41    },
42    /// The session cannot be written as JSON (non-UTF-8 `cwd`, non-finite timestamp).
43    #[error("cannot encode record {id}: {source}")]
44    Encode {
45        id: SessionId,
46        #[source]
47        source: RecordError,
48    },
49}
50
51fn io_error(path: &Path) -> impl FnOnce(io::Error) -> StoreError + '_ {
52    move |source| StoreError::Io {
53        path: path.to_owned(),
54        source,
55    }
56}
57
58fn file_name(path: &Path) -> String {
59    path.file_name()
60        .map(|name| name.to_string_lossy().into_owned())
61        .unwrap_or_default()
62}

64#[derive(Clone, Debug, PartialEq, Eq)]
65pub struct SessionStore {
66    dir: PathBuf,
67}

69impl SessionStore {
70    pub fn new(dir: impl Into<PathBuf>) -> Self {
71        Self { dir: dir.into() }
72    }
73
74    pub fn dir(&self) -> &Path {
75        &self.dir
76    }
77
78    pub fn path(&self, id: &SessionId) -> PathBuf {
79        self.dir.join(format!("{id}{RECORD_SUFFIX}"))
80    }
81
82    /// Atomic write: `.<id>.XXXXXXXX.tmp` in the same directory (mode 0600), then rename over
83    /// `<id>.json`. The temp file is removed if anything fails before the rename.
84    pub fn save(&self, session: &Session) -> Result<(), StoreError> {
85        let payload = session.to_json().map_err(|source| StoreError::Encode {
86            id: session.id.clone(),
87            source,
88        })?;
89        std::fs::create_dir_all(&self.dir).map_err(io_error(&self.dir))?;
90        let prefix = format!(".{}.", session.id);
91        let mut temp = tempfile::Builder::new()
92            .prefix(&prefix)
93            .suffix(TEMP_SUFFIX)
94            .rand_bytes(8)
95            .tempfile_in(&self.dir)
96            .map_err(io_error(&self.dir))?;
97        temp.write_all(payload.as_bytes())
98            .map_err(io_error(temp.path()))?;
99        temp.as_file().sync_all().map_err(io_error(temp.path()))?;
100        let target = self.path(&session.id);
101        temp.persist(&target)
102            .map_err(|error| io_error(&target)(error.error))?;
103        // The rename itself is durable only once the directory is synced.
104        std::fs::File::open(&self.dir)
105            .and_then(|dir| dir.sync_all())
106            .map_err(io_error(&self.dir))?;
107        Ok(())
108    }
109
110    /// One record by id; `None` if there is no such file.
111    pub fn load(&self, id: &SessionId) -> Result<Option<Session>, StoreError> {
112        let path = self.path(id);
113        match read(&path, id) {
114            Ok(session) => Ok(Some(session)),
115            Err(ReadError::Io(error)) if error.kind() == io::ErrorKind::NotFound => Ok(None),
116            Err(source) => Err(StoreError::Unreadable {
117                file: file_name(&path),
118                source,
119            }),
120        }
121    }
122
123    /// Removes the record file. `Ok(false)` when there was none.
124    pub fn delete(&self, id: &SessionId) -> Result<bool, StoreError> {
125        let path = self.path(id);
126        match std::fs::remove_file(&path) {
127            Ok(()) => Ok(true),
128            Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(false),
129            Err(source) => Err(StoreError::Io { path, source }),
130        }
131    }
132
133    /// Every record, sorted by `created_at` then id. A missing directory is an empty store. A
134    /// record deleted between the directory scan and its read is skipped; any other unreadable
135    /// record fails the scan by name. Entries that are not `<name>.json` regular files (temp
136    /// files, hidden files, directories) are ignored.
137    pub fn all(&self) -> Result<Vec<Session>, StoreError> {
138        let entries = match std::fs::read_dir(&self.dir) {
139            Ok(entries) => entries,
140            Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(Vec::new()),
141            Err(source) => return Err(io_error(&self.dir)(source)),
142        };
143        let mut sessions = Vec::new();
144        for entry in entries {
145            let entry = entry.map_err(io_error(&self.dir))?;
146            let name = entry.file_name();
147            let Some(name) = name.to_str() else { continue };
148            if name.starts_with('.') || !name.ends_with(RECORD_SUFFIX) {
149                continue;
150            }
151            let path = entry.path();
152            // Only a record file can be a record: a directory, FIFO or socket named `x.json`
153            // is skipped (a FIFO would block `open` forever), and one that vanished since the
154            // scan is a deletion that raced us.
155            let Ok(id) = name[..name.len() - RECORD_SUFFIX.len()].parse::<SessionId>() else {
156                continue;
157            };
158            match std::fs::metadata(&path) {
159                Ok(meta) if meta.is_file() => {}
160                Ok(_) => continue,
161                Err(error) if error.kind() == io::ErrorKind::NotFound => continue,
162                Err(source) => {
163                    return Err(StoreError::Unreadable {
164                        file: name.to_owned(),
165                        source: ReadError::Io(source),
166                    });
167                }
168            }
169            match read(&path, &id) {
170                Ok(session) => sessions.push(session),
171                Err(ReadError::Io(error)) if error.kind() == io::ErrorKind::NotFound => {}
172                Err(source) => {
173                    return Err(StoreError::Unreadable {
174                        file: name.to_owned(),
175                        source,
176                    });
177                }
178            }
179        }
180        sessions.sort_by(|a, b| {
181            a.created_at
182                .total_cmp(&b.created_at)
183                .then_with(|| a.id.cmp(&b.id))
184        });
185        Ok(sessions)
186    }
187}

189/// Reads the record at `path`, which must be a regular file whose record carries `id`.
190fn read(path: &Path, id: &SessionId) -> Result<Session, ReadError> {
191    if !std::fs::metadata(path)?.is_file() {
192        return Err(ReadError::NotAFile);
193    }
194    let bytes = std::fs::read(path)?;
195    let text = std::str::from_utf8(&bytes).map_err(ReadError::Utf8)?;
196    let session = Session::from_json(text)?;
197    if session.id != *id {
198        return Err(ReadError::IdMismatch {
199            found: session.id.to_string(),
200        });
201    }
202    Ok(session)
203}

410    fn non_regular_files_are_skipped_without_blocking() {
411        let (_dir, store) = store();
412        store.save(&session("a", 1.0)).unwrap();
413        let fifo = store.dir().join("pipe.json");
414        assert!(
415            std::process::Command::new("mkfifo")
416                .arg(&fifo)
417                .status()
418                .unwrap()
419                .success()
420        );
421        let (tx, rx) = std::sync::mpsc::channel();
422        let scanner = store.clone();
423        std::thread::spawn(move || {
424            let ids: Vec<String> = scanner
425                .all()
426                .unwrap()
427                .iter()
428                .map(|s| s.id.to_string())
429                .collect();
430            let by_name = scanner.load(&sid("pipe"));
431            tx.send((ids, by_name.is_err())).unwrap();
432        });
433        let (ids, load_failed) = rx
434            .recv_timeout(std::time::Duration::from_secs(2))
435            .expect("the scan must not block on a FIFO");
436        assert_eq!(ids, ["a"]);
437        assert!(load_failed, "a named non-regular file is unreadable");
438    }
