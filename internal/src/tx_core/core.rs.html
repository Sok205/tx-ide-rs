193        self.parent
194            .as_ref()

196            && self
197                .tag
198                .as_ref()
199                .is_none_or(|tag| session.tags.contains(tag))

266/// `env` keys are shell-style names (`-e KEY=VALUE` cannot carry `=` or NUL; an empty key is
267/// accepted by tmux but reaches no process).
268fn validate_env(env: &[(String, String)]) -> Result<(), CoreError> {
269    for (key, value) in env {
270        let key_ok = !key.is_empty()
271            && key
272                .bytes()
273                .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-' || b == b'.');
274        if !key_ok || value.contains('\0') {
275            return Err(CoreError::InvalidEnvKey(key.clone()));
276        }
277    }
278    Ok(())
279}

281/// tmux's ways of saying the session we just made is already gone: no server behind the socket,
282/// the session is not there (`can't find session` from session-typed targets, `no such session`
283/// from the pane-typed `set-option` target), or the server quit while answering (`exit-empty`
284/// after the last session ended, which is ours). Anything else is an unknown answer, never read
285/// as "exited".
286fn session_vanished(stderr: &str) -> bool {
287    crate::tmux::is_no_server(stderr)
288        || stderr.starts_with("can't find session")
289        || stderr.starts_with("no such session")
290        || crate::tmux::server_quit(stderr)
291}

293/// `env[key] = value` for each pair: replaces in place, else appends.
294fn merge_env(env: &mut Vec<(String, String)>, layer: &[(String, String)]) {
295    for (key, value) in layer {
296        match env.iter_mut().find(|(existing, _)| existing == key) {
297            Some(slot) => slot.1.clone_from(value),
298            None => env.push((key.clone(), value.clone())),
299        }
300    }
301}

338    /// Checks the socket path, creates the state layout, writes a default `tmux.conf` if none.
339    /// Does not start tmux.
340    pub fn open(config: CoreConfig) -> Result<Self, CoreError> {
341        config.check_socket_path()?;
342        let state = &config.state_dir;
343        for sub in [SESSIONS_DIR, LOCKS_DIR, PROFILES_DIR] {
344            let dir = state.join(sub);
345            std::fs::create_dir_all(&dir).map_err(store_io(&dir))?;
346        }
347        let conf = config.tmux_conf_path();
348        if !conf.exists() {
349            std::fs::write(&conf, DEFAULT_TMUX_CONF).map_err(store_io(&conf))?;
350        }
351        Ok(Self {
352            store: SessionStore::new(state.join(SESSIONS_DIR)),
353            tmux: Tmux::new(TMUX_BINARY, config.socket_path(), conf),
354            events: EventLog::new(state.join(LOG_FILE)),
355            engines: EngineRegistry::new(),
356            profiles: Profiles::new(config.profile_dirs.clone(), config.skill_dirs.clone()),
357            locks: Locks::new(state.join(LOCKS_DIR)),
358            sinks: Vec::new(),
359            host: Default::default(),
360            config,
361        })
362    }

385    /// The log line first (the file never lags the sinks), then every sink in order. Runs only
386    /// once the operation has released its locks, so a sink may use another core on the same
387    /// state dir. A sink that panics is contained; the operation already happened.
388    fn emit(&self, events: Vec<Event>) {
389        for event in &events {
390            // The log is an operation trail, not the record; a failed append must not fail the
391            // operation that already happened.
392            let _ = self.events.append(event);
393            for sink in &self.sinks {
394                let _ = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
395                    sink.on_event(event);
396                }));
397            }
398        }
399    }

419    /// An id first, then a name: the `Running` record with it, else the most recently created.
420    pub fn resolve(&self, name_or_id: &str) -> Result<SessionId, CoreError> {
421        if let Ok(id) = name_or_id.parse::<SessionId>()
422            && self.store.load(&id)?.is_some()
423        {
424            return Ok(id);
425        }
426        let mut named: Vec<Session> = self
427            .store
428            .all()?
429            .into_iter()
430            .filter(|session| session.name == name_or_id)
431            .collect();
432        if let Some(running) = named
433            .iter()
434            .find(|session| session.lifecycle == Lifecycle::Running)
435        {
436            return Ok(running.id.clone());
437        }
438        named
439            .pop()
440            .map(|session| session.id)
441            .ok_or_else(|| CoreError::NotFound(name_or_id.to_owned()))
442    }

452    fn create_locked(&self, opts: CreateOptions) -> Result<(SessionId, Vec<Event>), CoreError> {
453        validate_name(&opts.name)?;
454        validate_env(&opts.env)?;
455        if !opts.cwd.is_dir() {
456            return Err(CoreError::NotADirectory(opts.cwd));
457        }
458        // The process runs in the real directory; the record and every workspace path say so.
459        let cwd = std::fs::canonicalize(&opts.cwd).map_err(store_io(&opts.cwd))?;
460        let opts = CreateOptions { cwd, ..opts };
461        // Pure checks before the lock: an unknown engine or role never touches state.
462        let (engine, roles) = match &opts.launch {
463            LaunchOptions::Agent {
464                engine, profile, ..
465            } => {
466                let adapter = self
467                    .engines
468                    .get(engine)
469                    .ok_or_else(|| CoreError::UnknownEngine(engine.clone()))?;
470                (
471                    Some((adapter, self.profiles.resolve(profile)?)),
472                    profile.clone(),
473                )
474            }
475            LaunchOptions::Process { program, .. } => {
476                if program.is_empty() {
477                    return Err(CoreError::InvalidName(program.clone()));
478                }
479                (None, Vec::new())
480            }
481            #[allow(unreachable_patterns)]
482            _ => unreachable!("LaunchOptions variants are all handled"),
483        };
484
485        let _create = self.locks.create().map_err(CoreError::Lock)?;
486        if self.name_in_use(&opts.name, None)? {
487            return Err(CoreError::NameTaken(opts.name));
488        }
489        let id = SessionId::mint();
490        let profile_file = self
491            .config
492            .state_dir
493            .join(PROFILES_DIR)
494            .join(format!("{id}.md"));
495        let cleanup_profile = || {
496            let _ = std::fs::remove_file(&profile_file);
497        };
498
499        let mut env = vec![
500            (SESSION_ID_ENV.to_owned(), id.to_string()),
501            (
502                TMUX_SOCKET_ENV.to_owned(),
503                self.tmux.socket().to_string_lossy().into_owned(),
504            ),
505            (
506                STATE_DIR_ENV.to_owned(),
507                self.config.state_dir.to_string_lossy().into_owned(),
508            ),
509        ];
510        merge_env(&mut env, &self.config.env);
511        merge_env(&mut env, &opts.env);
512
513        let (program, args, launch) = match (&opts.launch, engine) {
514            (
515                LaunchOptions::Agent {
516                    engine,
517                    model,
518                    effort,
519                    initial_prompt,
520                    resume,
521                    ..
522                },
523                Some((adapter, resolved)),
524            ) => {
525                let profile =
526                    Profile::write(resolved, &profile_file).map_err(store_io(&profile_file))?;
527                let request = LaunchRequest {
528                    cwd: &opts.cwd,
529                    profile: &profile,
530                    model: model.as_deref(),
531                    effort: *effort,
532                    initial_prompt: initial_prompt.as_deref(),
533                    resume: resume.as_deref(),
534                };
535                let plan = adapter
536                    .launch(&request)
537                    .inspect_err(|_| cleanup_profile())?;
538                apply_workspace(&opts.cwd, &plan.workspace).inspect_err(|_| cleanup_profile())?;
539                validate_env(&plan.env).inspect_err(|_| cleanup_profile())?;
540                merge_env(&mut env, &plan.env);
541                if plan.program.is_empty() {
542                    cleanup_profile();
543                    return Err(CoreError::InvalidName(plan.program));
544                }
545                (
546                    plan.program,
547                    plan.args,
548                    Launch::Agent {
549                        engine: engine.clone(),
550                        profile: roles,
551                        model: model.clone(),
552                        effort: *effort,
553                        resume: resume.clone(),
554                    },
555                )
556            }
557            (LaunchOptions::Process { program, args, .. }, _) => (
558                program.clone(),
559                args.clone(),
560                Launch::Process {
561                    program: program.clone(),
562                    args: args.clone(),
563                },
564            ),
565            (LaunchOptions::Agent { .. }, None) => unreachable!("engine resolved above"),
566        };
567
568        if let Err(error) = self
569            .tmux
570            .new_session(&opts.name, &opts.cwd, &program, &args, &env)
571        {
572            cleanup_profile();
573            return Err(match error {
574                TmuxError::Failed { stderr, .. } if stderr.starts_with("duplicate session") => {
575                    CoreError::NameTaken(opts.name)
576                }
577                other => CoreError::Tmux(other),
578            });
579        }
580        let undo = |error: CoreError| {
581            let _ = self.tmux.kill_session(&exact(&opts.name));
582            cleanup_profile();
583            error
584        };
585        // A program that exits at once takes its session (and, if it was the last one, the
586        // server) with it before `@tx-id` can be set. That is a session that ran and exited,
587        // not a failure to create one: the record says so.
588        let exited_at_once = match self.tmux.set_tx_id(&opts.name, &id) {
589            Ok(()) => false,
590            Err(TmuxError::Failed { stderr, .. }) if session_vanished(&stderr) => true,
591            Err(error) => return Err(undo(CoreError::Tmux(error))),
592        };
593        let created_at = now();
594        let session = Session {
595            id: id.clone(),
596            name: opts.name.clone(),
597            parent: opts.parent,
598            tags: opts.tags,
599            cwd: opts.cwd,
600            created_at,
601            ended_at: exited_at_once.then_some(created_at),
602            lifecycle: if exited_at_once {
603                Lifecycle::Exited
604            } else {
605                Lifecycle::Running
606            },
607            launch,
608            ext: std::collections::BTreeMap::new(),
609        };
610        if let Err(error) = self.store.save(&session) {
611            return Err(undo(CoreError::Store(error)));
612        }
613        let mut events = vec![event(EventType::Created, &session, None, "")];
614        if exited_at_once {
615            events.push(event(
616                EventType::Exited,
617                &session,
618                None,
619                "exited before registration",
620            ));
621        }
622        Ok((id, events))
623    }

636    pub fn rename_session(&self, id: &SessionId, name: &str) -> Result<Session, CoreError> {
637        let (session, events) = self.rename_locked(id, name)?;
638        self.emit(events);

642    fn rename_locked(
643        &self,
644        id: &SessionId,
645        name: &str,
646    ) -> Result<(Session, Vec<Event>), CoreError> {
647        validate_name(name)?;
648        let _create = self.locks.create().map_err(CoreError::Lock)?;
649        let _session = self.locks.session(id).map_err(CoreError::Lock)?;
650        let mut session = self.get_session(id)?;
651        if session.name == name {
652            return Ok((session, Vec::new()));
653        }
654        if self.name_in_use(name, Some(id))? {
655            return Err(CoreError::NameTaken(name.to_owned()));
656        }
657        if session.lifecycle == Lifecycle::Running
658            && let Some(live) = self.tmux.is_live(id)?
659        {
660            match self.tmux.rename_session(&live.session_target, name) {
661                Ok(()) => {}
662                Err(TmuxError::Failed { stderr, .. })
663                    if stderr.starts_with("duplicate session") =>
664                {
665                    return Err(CoreError::NameTaken(name.to_owned()));
666                }
667                Err(error) => return Err(CoreError::Tmux(error)),
668            }
669        }
670        let old = std::mem::replace(&mut session.name, name.to_owned());
671        self.store.save(&session)?;
672        let renamed = event(
673            EventType::Renamed,
674            &session,
675            None,
676            format!("{old} -> {name}"),
677        );
678        Ok((session, vec![renamed]))
679    }

681    /// Runs `f` on `ext[plugin]` (`Null` when absent) under the session lock and saves the
682    /// record if the value changed; leaving it `Null` removes the key. Returns whether a write
683    /// happened. `f` must not call back into the core (the lock is held).
684    pub fn update_ext(
685        &self,
686        id: &SessionId,
687        plugin: &str,
688        f: impl FnOnce(&mut serde_json::Value),
689    ) -> Result<bool, CoreError> {
690        let _session = self.locks.session(id).map_err(CoreError::Lock)?;
691        let mut session = self.get_session(id)?;
692        let before = session
693            .ext
694            .get(plugin)
695            .cloned()
696            .unwrap_or(serde_json::Value::Null);
697        let mut value = before.clone();
698        f(&mut value);
699        if value == before {
700            return Ok(false);
701        }
702        if value.is_null() {
703            session.ext.remove(plugin);
704        } else {
705            session.ext.insert(plugin.to_owned(), value);
706        }
707        self.store.save(&session)?;
708        Ok(true)
709    }

728    /// One `list-sessions`; a `Running` record without a live `@tx-id` becomes `Exited`.
729    pub fn refresh_session(&self, id: &SessionId) -> Result<RefreshResult, CoreError> {
730        let (result, exited) = {
731            let _session = self.locks.session(id).map_err(CoreError::Lock)?;
732            let session = self.get_session(id)?;
733            if session.lifecycle.is_terminal() || self.tmux.list_live()?.contains(id) {
734                (RefreshResult::Unchanged(session), None)
735            } else {
736                let (session, exited) = self.mark_exited(session)?;
737                (RefreshResult::Exited(session), Some(exited))
738            }
739        };
740        self.emit(exited.into_iter().collect());
741        Ok(result)
742    }

744    /// One `list-sessions` for every record; returns the records that changed. Records are
745    /// scanned before tmux is listed: a record exists only after its tmux session does, so a
746    /// record missing from a later listing has truly gone (the other order would mark a session
747    /// created between the two calls as exited).
748    pub fn refresh_all(&self) -> Result<Vec<Session>, CoreError> {
749        let records = self.store.all()?;
750        let live = self.tmux.list_live()?;
751        let mut changed = Vec::new();
752        for session in records {
753            if session.lifecycle != Lifecycle::Running || live.contains(&session.id) {
754                continue;
755            }
756            let exited = {
757                let _lock = self.locks.session(&session.id).map_err(CoreError::Lock)?;
758                // Re-load under the lock: another process may have stopped it meanwhile.
759                match self.store.load(&session.id)? {
760                    Some(current) if current.lifecycle == Lifecycle::Running => {
761                        Some(self.mark_exited(current)?)
762                    }
763                    _ => None,
764                }
765            };
766            // Emitted per record, right after its lock is released: a later failure cannot
767            // swallow an exit that was already saved, and sinks see each exit as it happens.
768            if let Some((session, exited)) = exited {
769                self.emit(vec![exited]);
770                changed.push(session);
771            }
772        }
773        Ok(changed)
774    }

776    /// Saves the record as `Exited`; the caller emits the returned event once its lock is gone.
777    fn mark_exited(&self, mut session: Session) -> Result<(Session, Event), CoreError> {
778        session.lifecycle = Lifecycle::Exited;
779        session.ended_at = Some(now());
780        self.store.save(&session)?;
781        let exited = event(EventType::Exited, &session, None, "");
782        Ok((session, exited))
783    }

785    pub fn get_execution_location(&self, id: &SessionId) -> Result<ExecutionLocation, CoreError> {
786        if self.get_session(id)?.lifecycle.is_terminal() {
787            return Err(CoreError::NotRunning(id.clone()));
788        }
789        let live = self.live_or_not_running(id)?;
790        Ok(ExecutionLocation {
791            socket: self.tmux.socket().to_owned(),
792            session_target: live.session_target,
793            pane_target: live.pane_target,

797    /// The recipient's engine frames the message when it is registered here; a `Process`
798    /// session, or an agent whose engine this core does not know, gets the default envelope.
799    fn envelope_for(&self, recipient: &Session, sender: Option<&str>, text: &str) -> String {
800        let adapter = match &recipient.launch {
801            Launch::Agent { engine, .. } => self.engines.get(engine),
802            Launch::Process { .. } => None,
803        };
804        match adapter {
805            Some(adapter) => adapter.envelope(sender, text),
806            None => build_envelope(sender, text),
807        }
808    }

811        self.tmux
812            .is_live(id)?
813            .ok_or_else(|| CoreError::NotRunning(id.clone()))
814    }

816    /// `kill-session`, then `Stopped`. A record whose session had already vanished becomes
817    /// `Exited` instead (what happened, not what was asked).
818    pub fn stop_session(&self, id: &SessionId) -> Result<Session, CoreError> {
819        let (session, ended) = {
820            let _session = self.locks.session(id).map_err(CoreError::Lock)?;
821            let mut session = self.get_session(id)?;
822            if session.lifecycle.is_terminal() {
823                return Err(CoreError::NotRunning(id.clone()));
824            }
825            match self.tmux.is_live(id)? {
826                None => self.mark_exited(session)?,
827                Some(live) => {
828                    self.tmux.kill_session(&live.session_target)?;
829                    session.lifecycle = Lifecycle::Stopped;
830                    session.ended_at = Some(now());
831                    self.store.save(&session)?;
832                    let stopped = event(EventType::Stopped, &session, None, "");
833                    (session, stopped)
834                }
835            }
836        };
837        self.emit(vec![ended]);
838        Ok(session)
839    }

841    /// Deletes the record (and its profile file). The tmux session, if any, keeps running and is
842    /// no longer managed. `Ok(false)` when there was no record.
843    pub fn forget_session(&self, id: &SessionId) -> Result<bool, CoreError> {
844        let (existed, name) = {
845            let _session = self.locks.session(id).map_err(CoreError::Lock)?;
846            let name = self
847                .store
848                .load(id)
849                .ok()
850                .flatten()
851                .map(|session| session.name);
852            let existed = self.store.delete(id)?;
853            let _ = std::fs::remove_file(
854                self.config
855                    .state_dir
856                    .join(PROFILES_DIR)
857                    .join(format!("{id}.md")),
858            );
859            (existed, name)
860        };
861        if existed {
862            self.emit(vec![Event::new(
863                EventType::Forgotten,
864                id.clone(),
865                name.unwrap_or_default(),
866                None,
867                "",
868            )]);
869        }
870        Ok(existed)
871    }

875    pub fn send_message(
876        &self,
877        to: &SessionId,
878        text: &str,
879        opts: SendOptions,
880    ) -> Result<DeliveryReceipt, CoreError> {
881        let sender_name = match &opts.sender {
882            Some(sender) => Some(self.get_session(sender)?.name),
883            None => None,
884        };
885        let (receipt, sent) = {
886            let _session = self.locks.session(to).map_err(CoreError::Lock)?;
887            let recipient = self.get_session(to)?;
888            if recipient.lifecycle != Lifecycle::Running {
889                return Err(CoreError::NotRunning(to.clone()));
890            }
891            let live = self.live_or_not_running(to)?;
892            let envelope = self.envelope_for(&recipient, sender_name.as_deref(), text);
893            deliver(&self.tmux, &live.pane_target, &envelope, DELIVERY_PAUSE)?;
894            let submitted_at = now();
895            let sent = event(
896                EventType::MessageSent,
897                &recipient,
898                opts.sender.clone(),
899                format!("from {}", sender_name.as_deref().unwrap_or("user")),
900            );
901            (
902                DeliveryReceipt {
903                    recipient: to.clone(),
904                    pane: live.pane_target,
905                    submitted_at,
906                },
907                sent,
908            )
909        };
910        self.emit(vec![sent]);
911        Ok(receipt)
912    }
